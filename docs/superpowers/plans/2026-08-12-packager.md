# Packager Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a `packager` CLI that previews a local `.deb` or `.rpm`, writes a PKGBUILD, runs `makepkg`, optionally `pacman -U`, and later reminds the user when extra or the AUR has a native package.

**Architecture:** A Go CLI that shells out to `bsdtar`, `readelf`, `pkgfile`, `makepkg`, and `pacman`. Each spec unit is an `internal/` package with a narrow API. The PKGBUILD is the conversion artifact. Tests inject fakes for pacman/AUR; default `go test ./...` never runs `pacman -U` and never needs root.

**Tech Stack:** Go 1.22+, standard library only (`flag`, `os/exec`, `encoding/json`, `archive/tar`, `compress/gzip`, `net/http` for AUR RPC). External binaries as in the spec. Module path: `packager`. Binary: `cmd/packager`.

**Spec:** `docs/superpowers/specs/2026-08-12-packager-design.md`

## Global Constraints

- Temporary utility, not a package manager or AUR helper.
- `.deb` and `.rpm` → pacman only. No reverse conversion.
- Default verb: preview → convert → install. Skip install with `convert`. Stop after PKGBUILD with `scaffold`.
- Native extra/AUR match is a hard stop unless `--force`.
- Offline extra/AUR lookup is a hard stop unless `--force` (preview cannot claim “no native package”).
- Vendor scripts are never executed during convert. Packaged as `.install` only with `--allow-scripts`. No rewriting of `update-alternatives` / `update-rc.d` / `debconf`.
- Do not rewrite `/usr/lib/x86_64-linux-gnu` or `/usr/lib64` into Arch paths. Detect and warn only.
- Unmapped depends are omitted from the PKGBUILD `depends` array.
- Do not pass `-s` to `makepkg`.
- `forget` always asks about `pacman -R`; default is no; if remove fails, do not delete JSON.
- State: `~/.local/share/packager/installed/<pkgname>.json` written as the original user when invoked via sudo (`SUDO_USER`).
- PKGBUILD sets `packager = "packager <packager@local>"`.
- `pkgname`: lowercased, `_` → `-`. `pkgrel` is always `1`. `amd64`/`x86_64` → `x86_64`; `arm64`/`aarch64` → `aarch64`; anything else is wrong arch (stop).
- Workdir default: `~/.cache/packager/<pkgname>-<pkgver>/`. Never auto-delete.
- Root only for `pacman -U` and optional `pacman -R`. Stream pacman output. `--noconfirm` only with `--yes` on the action the user asked for.
- No vendor blobs in the repo. Generate tiny packages in the test harness.
- Default tests: no root, no `pacman -U`. `@system` tests are `-tags=system` and opt-in.
- stdlib only. Do not add cobra, debtap, alien, or fpm.

```go
package ident

import (
	"fmt"
	"strings"
	"unicode"
)

type Format string

const (
	FormatDeb Format = "deb"
	FormatRPM Format = "rpm"
	PkgRel           = "1"
	PackagerField    = "packager <packager@local>"
)

type Arch string

const (
	ArchX86_64  Arch = "x86_64"
	ArchAArch64 Arch = "aarch64"
)

func NormalizeName(raw string) string {
	s := strings.ToLower(strings.TrimSpace(raw))
	s = strings.ReplaceAll(s, "_", "-")
	return s
}

func NormalizeVer(raw string) string {
	var b strings.Builder
	for _, r := range raw {
		switch {
		case r == ':':
			b.WriteByte('.')
		case r == '/':
			b.WriteByte('_')
		case unicode.IsPrint(r):
			b.WriteRune(r)
		}
	}
	return b.String()
}

func MapArch(raw string) (Arch, error) {
	switch strings.ToLower(strings.TrimSpace(raw)) {
	case "amd64", "x86_64":
		return ArchX86_64, nil
	case "arm64", "aarch64":
		return ArchAArch64, nil
	default:
		return "", fmt.Errorf("wrong arch: %s", raw)
	}
}
```

```go
func TestSystemInstallForget(t *testing.T) {
	if os.Geteuid() != 0 {
		t.Skip("rerun as root: go test ./internal/cli -tags=system")
	}
	state.SetDataDirForTest(t.TempDir())
	t.Cleanup(func() {
		state.SetDataDirForTest("")
		_ = exec.Command("pacman", "-R", "--noconfirm", "packager-systest").Run()
	})
	LookupClient = noneLookup()
	OwnedBy = func(string) (string, bool, error) { return "", false, nil }
	MapResolver = noneRes{}
	t.Cleanup(func() { LookupClient = lookup.Client{}; OwnedBy = nil; MapResolver = nil; ForgetRemove = nil })

	deb := filepath.Join(t.TempDir(), "packager-systest_1.0_amd64.deb")
	if err := testpkg.WriteDeb(deb, testpkg.DebSpec{
		Name: "packager-systest", Version: "1.0", Arch: "amd64",
		Files: map[string][]byte{"./usr/share/packager-systest/README": []byte("ok\n")},
	}); err != nil {
		t.Fatal(err)
	}
	wd := t.TempDir()
	if code := Run([]string{"install", "-y", deb, "--workdir", wd}); code != 0 {
		t.Fatalf("install exit %d", code)
	}
	if out, err := exec.Command("pacman", "-Q", "packager-systest").CombinedOutput(); err != nil {
		t.Fatalf("pacman -Q: %s %v", out, err)
	}

	ForgetRemove = func(string) bool { return false }
	if code := Run([]string{"forget", "packager-systest"}); code != 0 {
		t.Fatalf("forget exit %d", code)
	}
	if _, err := state.Read("packager-systest"); err == nil {
		t.Fatal("json should be gone")
	}
	if out, err := exec.Command("pacman", "-Q", "packager-systest").CombinedOutput(); err != nil {
		t.Fatalf("package should still be installed: %s %v", out, err)
	}

	if err := state.Write(state.Record{PkgName: "packager-systest", PkgVer: "1.0", SourceName: filepath.Base(deb)}); err != nil {
		t.Fatal(err)
	}
	ForgetRemove = func(string) bool { return true }
	if code := Run([]string{"forget", "packager-systest"}); code != 0 {
		t.Fatalf("forget+remove exit %d", code)
	}
	if err := exec.Command("pacman", "-Q", "packager-systest").Run(); err == nil {
		t.Fatal("package should be removed")
	}
}
```
